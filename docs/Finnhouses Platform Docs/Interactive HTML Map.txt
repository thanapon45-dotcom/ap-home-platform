// Labels

  .attr("class", d => "node-label" + (d.type === "doc" ? " title" : ""))
  .attr("text-anchor", "middle")
  .attr("y", d => d.type === "doc" ? 4 : 4)
  .selectAll("tspan").data(d => d.label.split("\n")).join("tspan")
  .attr("x", 0)
  .attr("dy", (d, i, nodes) => i === 0 ? `-${(nodes.length - 1) * 0.6}em` : "1.2em");

// Position
nodeGroup.attr("transform", d => `translate(${d.x},${d.y})`);

// Tooltip + highlight
const tooltip = d3.select("#tooltip");
nodeGroup
  .on("mouseenter", function(event, d) {
    tooltip.style("opacity", 1).html(`<b>${d.label.replace("\n", " ")}</b><br><br>${d.desc || ""}`);
    d3.select(this).select("rect").attr("stroke", "#37352f").attr("stroke-width", 2.5);
  })
  .on("mousemove", (event) => {
    tooltip.style("left", (event.pageX + 16) + "px").style("top", (event.pageY + 10) + "px");
  })
  .on("mouseleave", function() {
    tooltip.style("opacity", 0);
    d3.select(this).select("rect").attr("stroke", "#ffffff").attr("stroke-width", 1.5);
  })
  .on("click", function(event, d) {
    event.stopPropagation();
    highlight(d.id);
  });

let highlighted = null;
function highlight(id) {
  if (highlighted === id) { clearHighlight(); highlighted = null; return; }
  highlighted = id;

  // Find connected nodes
  const connected = new Set([id]);
  links.forEach(l => {
    if (l.source === id) connected.add(l.target);
    if (l.target === id) connected.add(l.source);
  });

  nodeGroup.select("rect").classed("dim", d => !connected.has(d.id));
  nodeGroup.select("text").classed("dim", d => !connected.has(d.id));
  link.classed("dim", l => !(connected.has(l.source) && connected.has(l.target)));
}
function clearHighlight() {
  nodeGroup.select("rect").classed("dim", false);
  nodeGroup.select("text").classed("dim", false);
  link.classed("dim", false);
}

svg.on("click", (event) => {
  if (event.target.tagName === "svg" || event.target.tagName === "rect") {
    clearHighlight();
    highlighted = null;
  }
});

// Simple drag
const drag = d3.drag()
  .on("start", function(event, d) {
    if (!event.active) return;
    d3.select(this).raise();
  })
  .on("drag", function(event, d) {
    d.x = event.x;
    d.y = event.y;
    d3.select(this).attr("transform", `translate(${d.x},${d.y})`);
    link
      .attr("x1", l => nodes.find(n => n.id === l.source).x)
      .attr("y1", l => nodes.find(n => n.id === l.source).y)
      .attr("x2", l => nodes.find(n => n.id === l.target).x)
      .attr("y2", l => nodes.find(n => n.id === l.target).y);
  });
nodeGroup.call(drag);

link
  .attr("x1", d => nodes.find(n => n.id === d.source).x)
  .attr("y1", d => nodes.find(n => n.id === d.source).y)
  .attr("x2", d => nodes.find(n => n.id === d.target).x)
  .attr("y2", d => nodes.find(n => n.id === d.target).y);
</script>
